--- p9 ---
二分查找 逻辑
思路同猜数字大小：
1.定义两个变量l和r；
2.用mid（猜的数）保存l和r的中间值；
3.假如小了，则把l变成mid+1（去除左半部）；
4.假如大了，则把r变成mid-1（去除右半部）；
5.假如对了，则返回mid；
6.假如l > r，则返回-1；
7.重复第2-6句

二分查找 案例
在数组长度为N（N<=100)的数组中，输入m个元素（有序），查找
k元素的位置(位置从1开始)，如果找不到k，输出-1.
案例输入：
5
11 22 33 44 55
22
案例输出：
2
--- p14 ---
二分查找 代码
                    int const N = 100;  //数组长度
int arr[N];  //声明数组
int m, k;    //m元素个数，k查找的元素
cin >> m;
for (int i = 0; i < m; i++) {
cin >> arr[i];
}
cin >> k;
//left起始下标，right最大下标，mid保存中间下标
int left = 0, right = m - 1, mid = 0;
while (left <= right) {  //当left小于等于right时查找
mid = (left + right) / 2;  //计算中间值mid
if (k == arr[mid])   //查找到k
{
cout << mid + 1 << endl;
break;
}
else if (k < arr[mid]) { //小于的情况，right挪到mid-1
right = mid - 1;
}
else if (k > arr[mid]) { //大于的情况，left挪到mid+1
left = mid + 1;
}
}
if (left > right)  //当left <right ,k元素不在数组中
{
cout << -1 << endl;
}

--- p23 ---
查找元素个数代码
                    int const N = 100;  //数组长度
int arr[N];  //声明数组
int m, k;    //m元素个数，k查找的元素
cin >> m;
for (int i = 0; i < m; i++) {
cin >> arr[i];
}
cin >> k;
//left起始下标，right最大下标，mid保存中间下标
int left = 0, right = m - 1, mid = 0,count = 0;
while (left <= right) {  //当left小于等于right时查找
mid = (left + right) / 2;  //计算中间值mid
if (k == arr[mid])   //查找到k
{
int index1 = mid-1,index2 = mid+1;
count = 1;
while (index1 >= 0 && arr[index1] == k) {//扫描左边
index1--;
count++;
}
while (index2 <= m-1 && arr[index2] == k){//扫描右边
index2++;
count++;
}
break;
}
else if (k < arr[mid]) { //小于的情况，right挪到mid-1
right = mid - 1;
}
else if (k > arr[mid]) { //大于的情况，left挪到mid+1
left = mid + 1;
}
}
cout << count << endl;

--- p30 ---
先查找，在修改  代码
double arrScore[20];
int n;
cin >> n;
for (int i = 0; i < n; i++)
{
cin >> arrScore[i];
}
double first = arrScore[0], last = arrScore[0];
int first_index = 0, last_index = 0;
for (int i = 0; i < n; i++) {
if (arrScore[i] < first) {
first = arrScore[i];
first_index = i;
}
if (arrScore[i] > last)
{
last = arrScore[i];
last_index = i;
}
}
swap(arrScore[first_index], arrScore[last_index]);
for (int i = 0; i < n; i++) {
cout << arrScore[i] << " ";
}

--- p37 ---
根据要求添加到数组 代码
int arr[100];
int n, k, p, m;
cin >> n;
for (int i = 0; i < n; i++)
{
cin >> arr[i];
}
cin >> k;
cin >> p >> m;
int index = n - 1;
if (p) {
while (index >= m - 1)
{
arr[index + 1] = arr[index];
index--;
}
arr[m-1] = k;
}
else {
while (index > m -1)
{
arr[index + 1] = arr[index];
index--;
}
arr[m] = k;
}
for (int i = 0; i < (n + 1); i++) {
cout << arr[i] << " ";
}

有序合并 案例
输入两个升序整数数组nums1和nums2 ,将nums1与nums2合并到nums3中，并且删除重复的数,
使得nums3依旧为-个升序数组。( 不能先合并再排序)
输入格式:两行,第一-行先输入一个整数N1(0< N1 < 100)代表nums1数组的元素个数,再输入N1个
整数代表nums1中的N1个元素值。第二行先输入一个整数N2(0<N2< 100)代表nums2数组的元
素个数,再输入N2个整数代表num2中的N2个元素值。
输出格式:一行,合并后的nums3数组。
输入用例:
6
 -3  2  5  7  8  12
8
0   1  2   3  10    13    18   20
输出用例:
-3  0  1  2  3  5  7  8  10  12  13  18   20
--- p47 ---
有序合并 代码
int n1, n2, arr1[100], arr2[100], arr3[200];
cin >> n1;
for (int i = 0; i < n1; i++)
{
cin >> arr1[i];
}
cin >> n2;
for (int i = 0; i < n2; i++)
{
cin >> arr2[i];
}
int index = 0,i=0,j=0;
for (; i < n1 && j < n2;) {
if (arr1[i] < arr2[j]) {
arr3[index++] = arr1[i++];
}
else if (arr1[i] > arr2[j]) {
arr3[index++] = arr2[j++];
}
else {
arr3[index++] = arr2[j++];
i++;
}
}
while (i < n1)
{
arr3[index++] = arr1[i++];
}
while (j < n2) {
arr3[index++] = arr2[j++];
}
for (int i = 0; i < index; i++)
{
cout << arr3[i] << " ";
}
